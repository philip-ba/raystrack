cmake_minimum_required(VERSION 3.21)
project(raystrack_native VERSION 0.1.0 LANGUAGES CXX)
include(CTest)

option(RAYSTRACK_EMBREE "Build the Embree CPU traversal adapter" OFF)
option(RAYSTRACK_VULKAN "Build the Vulkan hardware traversal adapter" OFF)

add_library(raystrack_native SHARED api.cpp builtin.cpp shortwave.cpp)
target_compile_features(raystrack_native PRIVATE cxx_std_20)
target_include_directories(raystrack_native PRIVATE include)
set_target_properties(raystrack_native PROPERTIES CXX_VISIBILITY_PRESET hidden)

if(BUILD_TESTING)
    add_executable(raystrack_scattering_test scattering_test.cpp)
    target_compile_features(raystrack_scattering_test PRIVATE cxx_std_20)
    target_include_directories(raystrack_scattering_test PRIVATE include)
    add_test(NAME raystrack_scattering COMMAND raystrack_scattering_test)
endif()

if(RAYSTRACK_EMBREE)
    add_subdirectory(embree)
    target_link_libraries(raystrack_native PRIVATE raystrack_embree)
    target_compile_definitions(raystrack_native PRIVATE RAYSTRACK_EMBREE)
endif()
if(RAYSTRACK_VULKAN)
    add_subdirectory(vulkan)
    target_link_libraries(raystrack_native PRIVATE raystrack_vulkan)
    target_compile_definitions(raystrack_native PRIVATE RAYSTRACK_VULKAN)
endif()
